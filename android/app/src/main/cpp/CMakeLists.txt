cmake_minimum_required(VERSION 3.22.1)
project(appmodules LANGUAGES C CXX)
set_property(GLOBAL PROPERTY JOB_POOLS ziv_compile=3 ziv_link=1)
set(CMAKE_JOB_POOL_COMPILE ziv_compile)
set(CMAKE_JOB_POOL_LINK ziv_link)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(PLZMA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../Vendor/PLzmaSDK" ABSOLUTE)
# Compile sources directly into the shared library so static codec registrars survive linking.
file(GLOB_RECURSE PLZMA_SOURCES CONFIGURE_DEPENDS "${PLZMA_ROOT}/src/*.c" "${PLZMA_ROOT}/src/*.cpp")
add_library(zipimageviewer_native SHARED seven_zip_jni.cpp ${PLZMA_SOURCES})
target_include_directories(zipimageviewer_native PRIVATE "${PLZMA_ROOT}")
target_compile_definitions(zipimageviewer_native PRIVATE LIBPLZMA_VERSION_BUILD=1448 LIBPLZMA_NO_C_BINDINGS=1 LIBPLZMA_NO_TAR=1)
target_compile_options(zipimageviewer_native PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fno-rtti>)
# Portable C decoder on both ABIs. Do not define Z7_LZMA_DEC_OPT without corresponding assembly.
if(ANDROID)
    target_link_options(zipimageviewer_native PRIVATE "-Wl,-z,max-page-size=16384")
else()
    # Host-only JNI smoke tests; the shipping build always uses the Android NDK.
    find_package(JNI REQUIRED COMPONENTS JVM)
    target_include_directories(zipimageviewer_native PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(zipimageviewer_native PRIVATE pthread)
endif()

option(ZIV_ENABLE_JXL "Build pinned JPEG XL codec" ${ANDROID})
if(ZIV_ENABLE_JXL)
    get_filename_component(REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
    set(ZIV_JXL_SOURCE_DIR "${REPOSITORY_ROOT}/android/vendor/libjxl" CACHE PATH "Pinned libjxl checkout")
    set(codec_scripts "${REPOSITORY_ROOT}/Vendor/JPEGXL/Scripts")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
        "${codec_scripts}/source-pins.sh" "${codec_scripts}/checkout-source.sh")
    execute_process(COMMAND "${codec_scripts}/checkout-source.sh" "${ZIV_JXL_SOURCE_DIR}" android --verify
        RESULT_VARIABLE codec_verification)
    if(NOT codec_verification STREQUAL "0")
        message(FATAL_ERROR "JPEG XL source verification failed. Run android/vendor/bootstrap-jpegxl.sh before building.")
    endif()
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
    set(CMAKE_POSITION_INDEPENDENT_CODE ON)
    foreach(feature TOOLS DEVTOOLS EXAMPLES BENCHMARK FUZZERS DOXYGEN MANPAGES JNI PLUGINS VIEWERS OPENEXR SJPEG TCMALLOC)
        set(JPEGXL_ENABLE_${feature} OFF CACHE BOOL "" FORCE)
    endforeach()
    set(JPEGXL_ENABLE_SKCMS ON CACHE BOOL "" FORCE)
    set(JPEGXL_ENABLE_TRANSCODE_JPEG ON CACHE BOOL "" FORCE)
    set(JPEGXL_ENABLE_BOXES ON CACHE BOOL "" FORCE)
    set(JPEGXL_BUNDLE_LIBPNG ON CACHE BOOL "" FORCE)
    set(JPEGXL_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
    set(PNG_SHARED OFF CACHE BOOL "" FORCE)
    set(PNG_STATIC ON CACHE BOOL "" FORCE)
    set(PNG_TESTS OFF CACHE BOOL "" FORCE)
    set(PNG_TOOLS OFF CACHE BOOL "" FORCE)
    set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    add_subdirectory("${ZIV_JXL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/libjxl" EXCLUDE_FROM_ALL)
    set(JXL_BRIDGE "${REPOSITORY_ROOT}/Vendor/JPEGXL/Sources/JPEGXLBridge")
    add_library(zipimageviewer_jxl SHARED jpeg_xl_jni.cpp "${JXL_BRIDGE}/JPEGXLBridge.cpp")
    target_include_directories(zipimageviewer_jxl PRIVATE "${JXL_BRIDGE}/include" "${ZIV_JXL_SOURCE_DIR}/third_party/libpng"
        "${CMAKE_CURRENT_BINARY_DIR}/libjxl/third_party/libpng" "${ZIV_JXL_SOURCE_DIR}/third_party/skcms")
    target_link_libraries(zipimageviewer_jxl PRIVATE jxl jxl_cms png_static)
    if(ANDROID)
        target_link_libraries(zipimageviewer_jxl PRIVATE jnigraphics)
        target_link_options(zipimageviewer_jxl PRIVATE "-Wl,-z,max-page-size=16384")
    else()
        target_include_directories(zipimageviewer_jxl PRIVATE ${JNI_INCLUDE_DIRS})
    endif()
endif()

# React Native owns appmodules; codec targets above remain independently linked.
if(ANDROID AND DEFINED REACT_ANDROID_DIR)
    add_subdirectory(react)
endif()
